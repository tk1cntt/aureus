"""Scenario tests: run strategies through full signal pipeline on real CSV data."""
import os
import pytest
import pandas as pd
from engine.state import SymbolState
from engine.signal_factory import create_signal_set
from engine.live_engine import execute_signals_for_candle
from engine.strategies.registry import StrategyRegistry
from engine.strategies.template import TemplateStrategy

# Load seed strategy configs (inline, matching seed_strategies.py)
TREND_CONT_CONFIG = { ... }  # Copy from seed_strategies.py
SESSION_SWEEP_CONFIG = { ... }
ORDER_FLOW_DOM_CONFIG = { ... }

class TestScenarioFullPipeline:
    """Run each seed strategy through the full signal pipeline on XAUUSD data."""

    def _run_replay(self, df: pd.DataFrame, strategy_configs: list, symbol: str = "XAUUSD") -> dict:
        """Replay helper: feed candles through signal+strategy pipeline."""
        import json
        cfg_path = os.path.join(os.path.dirname(__file__), "..", "symbols.json")
        sym_cfg = {}
        if os.path.exists(cfg_path):
            with open(cfg_path) as f:
                sym_cfg = json.load(f).get(symbol, {})

        signals = create_signal_set(symbol, sym_cfg)
        state = SymbolState(symbol)
        registry = StrategyRegistry()
        for cfg in strategy_configs:
            config = dict(cfg)
            config["name"] = config.get("name", cfg.get("name"))
            config["id"] = config.get("id", 0)
            registry.register(TemplateStrategy(config))

        all_accepted = []
        all_rejections = []
        window_size = 2000

        for i in range(len(df)):
            window = df.iloc[max(0, i - window_size + 1):i + 1]
            if len(window) < 5:
                continue

            state.transient_signals = {}
            execute_signals_for_candle(signals, window, state, symbol, redis_client=None)

            accepted = registry.evaluate_all(window, signals, state)
            rejections = registry.get_rejections(clear=True)

            all_accepted.extend(accepted)
            all_rejections.extend(rejections)

        return {
            "accepted": all_accepted,
            "rejections": all_rejections,
            "bar_count": len(df),
        }

    def test_xauusd_trend_cont_produces_results(self, xauusd_df):
        """TREND_CONT on XAUUSD real data should produce at least some triggers or rejections."""
        result = self._run_replay(xauusd_df, [TREND_CONT_CONFIG])
        total_events = len(result["accepted"]) + len(result["rejections"])
        assert total_events > 0, "TREND_CONT produced zero events on XAUUSD data"

    def test_xauusd_session_sweep_produces_results(self, xauusd_df):
        result = self._run_replay(xauusd_df, [SESSION_SWEEP_CONFIG])
        total_events = len(result["accepted"]) + len(result["rejections"])
        assert total_events > 0, "SESSION_SWEEP produced zero events on XAUUSD data"

    def test_xauusd_order_flow_dom_produces_results(self, xauusd_df):
        result = self._run_replay(xauusd_df, [ORDER_FLOW_DOM_CONFIG])
        total_events = len(result["accepted"]) + len(result["rejections"])
        assert total_events > 0, "ORDER_FLOW_DOM produced zero events on XAUUSD data"

    def test_reject_reasons_are_expected(self, xauusd_df):
        """All rejection reason_codes must be from the known set."""
        result = self._run_replay(xauusd_df, [TREND_CONT_CONFIG, SESSION_SWEEP_CONFIG, ORDER_FLOW_DOM_CONFIG])
        expected_reasons = {
            "CONTEXT_FILTER_FAILED", "SEQUENCE_NOT_MATCHED",
            "BACKFILL_NOT_READY", "VALIDATION_RULE_FAILED",
            "ORDER_PLAN_INVALID_SHAPE", "STRATEGY_EVALUATION_ERROR",
        }
        for rej in result["rejections"]:
            assert rej["reason_code"] in expected_reasons, f"Unexpected reason: {rej['reason_code']}"

    def test_all_strategies_combined_trigger_count(self, xauusd_df):
        """Running all 3 strategies together on 2000 bars should produce ≥1 accepted trigger."""
        result = self._run_replay(xauusd_df, [TREND_CONT_CONFIG, SESSION_SWEEP_CONFIG, ORDER_FLOW_DOM_CONFIG])
        # At minimum, strategies should produce SOME triggers or rejections
        # (pure acceptance depends on market conditions in the fixture data)
        assert len(result["accepted"]) + len(result["rejections"]) > 0

    def test_scenario_determinism(self, xauusd_df):
        """Two independent replays on same data produce identical results."""
        result_a = self._run_replay(xauusd_df, [TREND_CONT_CONFIG])
        result_b = self._run_replay(xauusd_df, [TREND_CONT_CONFIG])

        assert len(result_a["accepted"]) == len(result_b["accepted"])
        assert len(result_a["rejections"]) == len(result_b["rejections"])

        for a, b in zip(result_a["accepted"], result_b["accepted"]):
            assert a["t"] == b["t"]
            assert a["strategy"] == b["strategy"]
